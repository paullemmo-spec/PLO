/**
 * PLO BUDGET APP: Google Sheets web app
 * ------------------------------------------------------------------
 * Lives inside "Paul & Flo Running Sheet". Flo opens a link on her
 * iPhone (no Google account, no Claude) and every spend she adds is
 * written straight into the sheet, in the same style you use by hand:
 *
 *   • the spend goes in today's row, in the column she picks
 *       (Expenses / Misc Spendings / Groceries / Petrol / Income …)
 *   • a second spend that day turns the cell into  =SUM(70+45.26)
 *   • the note column gets  " / Coles (45.26)"   (repeats merge: Fitstop (54+54))
 *   • the cell keeps your "$"#,##0.00 formatting; Running Total formulas are never touched
 *
 * Tabs without Day/Date/Running Total columns (e.g. Houses) work as
 * simple lists: add/edit rows, with formatting copied from the row above.
 *
 * SETUP (one time, about 5 minutes):
 *  1. In the sheet: Extensions > Apps Script.
 *  2. Replace everything in Code.gs with this file.
 *  3. Click + > HTML, name it exactly  Index  and paste in Index.html.
 *  4. (Optional) Choose  setup  in the function list and press Run. Approve the prompt.
 *     The "App Settings" tab is also created automatically the first time the app opens.
 *  5. Deploy > New deployment > Web app.
 *        Execute as: Me      Who has access: Anyone
 *     Deploy, then copy the Web app URL.
 *  6. Send Flo the link. On her iPhone: open in Safari > Share > Add to Home Screen.
 *     No PIN: the first time, each phone just picks who is using it.
 *
 * KEEP THE LINK PRIVATE: anyone with it can open the app. To stop it working,
 * Deploy > Manage deployments > Archive (and make a new deployment later).
 * Show or hide sheets: in the app, gear > Settings > Sheets in the app
 * (or tick/untick them under "Tabs the app can see" in App Settings).
 *
 * After editing this code: Deploy > Manage deployments > Edit (pencil) >
 * Version: New version > Deploy. The link stays the same.
 */

/* =============================== ENTRY =============================== */

function doGet(e) {
  // Opens fast: no sheet reading here. It hands the page who is using it (from the link),
  // their saved look, and the last loaded figures (kept for a few hours), so the first screen
  // shows straight away and fresh figures follow a moment later.
  const prop = PropertiesService.getScriptProperties();
  const u = String((e && e.parameter && e.parameter.u) || '').toLowerCase();
  const who = u === 'owner' || u === 'paul' ? 'owner' : (u === 'partner' || u === 'flo' ? 'partner' : '');
  let prefs = {};
  try { if (who) prefs = JSON.parse(prop.getProperty('prefs_' + who) || '{}'); } catch (err) {}
  let url = '';
  try { url = ScriptApp.getService().getUrl(); } catch (err) {}
  const boot = { who: who, prefs: prefs, url: url, data: cacheGet_('quick') };
  const t = HtmlService.createTemplateFromFile('Index');
  t.boot = JSON.stringify(boot).replace(/</g, '\\u003c').replace(/\u2028/g, '\\u2028').replace(/\u2029/g, '\\u2029');
  return t.evaluate()
    .setTitle(prop.getProperty('appName') || 'PLO')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1, maximum-scale=1, viewport-fit=cover')
    .addMetaTag('apple-mobile-web-app-capable', 'yes')
    .addMetaTag('mobile-web-app-capable', 'yes');   // only Google's allowed meta tags (title comes from setTitle)
}

/** Any edit typed straight into the sheet clears the saved copy, so the app never opens on old figures. */
function onEdit() {
  try { CacheService.getScriptCache().remove('quick_n'); } catch (e) {}
}

/* Last loaded figures, kept in Google's cache (split into pieces, it holds 100KB per piece). */
function cachePut_(key, obj) {
  try {
    const c = CacheService.getScriptCache(), str = JSON.stringify(obj), size = 90000, n = Math.ceil(str.length / size);
    if (n > 40) return;
    const m = { [key + '_n']: String(n) };
    for (let i = 0; i < n; i++) m[key + '_' + i] = str.slice(i * size, (i + 1) * size);
    c.putAll(m, 21600);
  } catch (e) {}
}
function cacheGet_(key) {
  try {
    const c = CacheService.getScriptCache(), n = parseInt(c.get(key + '_n'), 10);
    if (!n) return null;
    const keys = []; for (let i = 0; i < n; i++) keys.push(key + '_' + i);
    const got = c.getAll(keys); let str = '';
    for (let i = 0; i < n; i++) { if (got[keys[i]] == null) return null; str += got[keys[i]]; }
    return JSON.parse(str);
  } catch (e) { return null; }
}

/** Each person's look (theme, light/dark, last account) saved against their name, not the phone. */
function savePrefs_(p) {
  const who = p.who === 'partner' ? 'partner' : (p.who === 'owner' ? 'owner' : '');
  if (!who) return { ok: false };
  const prop = PropertiesService.getScriptProperties();
  let cur = {};
  try { cur = JSON.parse(prop.getProperty('prefs_' + who) || '{}'); } catch (e) {}
  Object.keys(p.prefs || {}).forEach(k => { if (/^bt_/.test(k)) cur[k] = p.prefs[k]; });
  prop.setProperty('prefs_' + who, JSON.stringify(cur).slice(0, 8000));
  return { ok: true };
}

/** "haircut" → "Haircut", "  maccas  " → "Maccas"; words already typed with capitals are left alone. */
function tidyName_(n) {
  n = String(n || '').replace(/[()\/]/g, ' ').replace(/\s+/g, ' ').trim();
  return n.split(' ').map(w => w && w === w.toLowerCase() ? w.charAt(0).toUpperCase() + w.slice(1) : w).join(' ');
}

/** The GitHub-hosted app talks to the sheet through here (same actions as inside Google). */
function doPost(e) {
  let out;
  try {
    const req = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    out = { ok: true, data: api(req.action, '', req.payload || {}) };
  } catch (err) {
    out = { ok: false, error: String((err && err.message) || err) };
  }
  return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON);
}

/** Single entry point used by the app. */
function api(action, pin, payload) {
  payload = payload || {};
  if (action === 'savePrefs') return savePrefs_(payload);
  const ctx = auth_(pin);
  switch (action) {
    case 'bootstrap': return bootstrap_(ctx, payload);
    case 'addSpend':  return addSpend_(ctx, payload);
    case 'saveDay':   return saveDay_(ctx, payload);
    case 'editItems': return editItems_(ctx, payload);
    case 'moveItem':  return moveItem_(ctx, payload);
    case 'bulkChange': return bulkChange_(ctx, payload);
    case 'weekdayGet': return { cfg: weekdayCfg_(), found: weekendFix_(ctx, { dryRun: true }).moved };
    case 'weekdaySave': return weekdaySave_(ctx, payload);
    case 'createTab': return createTab_(ctx, payload);
    case 'renameTabs': return renameTabs_(ctx, payload);
    case 'planYear':  return planYear_(ctx, payload);
    case 'repeatList': return { items: repeatsOut_(ctx) };
    case 'nameAmount': return nameAmount_(ctx, payload);
    case 'repeatSave': return repeatSave_(ctx, payload);
    case 'repeatDelete': return repeatDelete_(ctx, payload);
    case 'addRow':    return addRow_(ctx, payload.tab, payload.values);
    case 'updateRow': return updateRow_(ctx, payload.tab, payload.row, payload.values, payload.expect);
    case 'deleteRow': return deleteRow_(ctx, payload.tab, payload.row, payload.expect);
    case 'admin':     return admin_(ctx);
    case 'saveAdmin': return saveAdmin_(ctx, payload);
    default: throw new Error('Unknown action');
  }
}

function setup() {
  const ss = SpreadsheetApp.getActive();
  let sh = ss.getSheetByName(SETTINGS_SHEET);
  if (!sh) {
    sh = ss.insertSheet(SETTINGS_SHEET);
    sh.getRange('A1').setValue('Budget app settings').setFontWeight('bold').setFontSize(14);
    sh.getRange('A2').setValue('Tick or untick to control the app. Changes apply next time the app refreshes.')
      .setFontColor('#666666');
    const rows = [
      [KEYS.ownerName, 'Paul'],
      [KEYS.partnerName, 'Flo'],
      [KEYS.appName, 'PLO']
    ];
    sh.getRange(4, 1, rows.length, 2).setValues(rows);
    sh.getRange(4, 1, rows.length, 1).setFontWeight('bold');

    sh.getRange(TAB_HEADER_ROW - 1, 1).setValue('Tabs the app can see').setFontWeight('bold').setFontSize(12);
    sh.getRange(TAB_HEADER_ROW, 1, 1, 3)
      .setValues([['Tab', 'Show in app', 'Header row (blank = auto)']])  // column D (Tab ID) is added by syncTabs_
      .setFontWeight('bold').setBackground('#F1F3F4');
    sh.setColumnWidth(1, 300); sh.setColumnWidth(2, 140); sh.setColumnWidth(3, 190);
    sh.getRange(TAB_HEADER_ROW - 1, 5).setValue('Extra logos (optional)').setFontWeight('bold').setFontSize(12);
    sh.getRange(TAB_HEADER_ROW, 5, 1, 2).setValues([['Word in the note', 'Brand website (e.g. bunnings.com.au)']])
      .setFontWeight('bold').setBackground('#F1F3F4');
    sh.setColumnWidth(5, 200); sh.setColumnWidth(6, 280);
  }
  syncTabs_(sh);
  const s = getSettings_();
  Logger.log('Setup done. ' + s.tabs.length + ' tabs listed in App Settings. No PIN needed: open the web app link.');
}


/** New tabs get listed automatically (switched on). */
/**
 * Keeps the "Tabs the app can see" list in step with your tabs, matched by Google's tab ID
 * (column D), so renaming a tab keeps its on/off switch and header-row setting, new tabs are
 * added (switched on), and deleted tabs drop off. Only writes when something actually changed.
 * The logo table in columns E:F is never touched.
 */
function syncTabs_(sh) {
  const ss = SpreadsheetApp.getActive();
  if (String(sh.getRange(TAB_HEADER_ROW, 4).getValue()) === '') {
    sh.getRange(TAB_HEADER_ROW, 4).setValue('Tab ID (automatic, don\u2019t edit)').setFontWeight('bold').setBackground('#F1F3F4').setFontColor('#999999');
    sh.setColumnWidth(4, 170);
  }
  const last = Math.max(sh.getLastRow(), TAB_HEADER_ROW);
  const n = last - TAB_HEADER_ROW;
  const rows = (n > 0 ? sh.getRange(TAB_HEADER_ROW + 1, 1, n, 4).getValues() : [])
    .filter(r => String(r[0]) !== '' || String(r[3]) !== '');
  const taken = {};
  const out = ss.getSheets().filter(t => t.getName() !== SETTINGS_SHEET).map(t => {
    const id = String(t.getSheetId()), name = t.getName();
    let k = rows.findIndex((r, i) => !taken[i] && String(r[3]) === id);
    if (k < 0) k = rows.findIndex((r, i) => !taken[i] && String(r[3]) === '' && String(r[0]) === name); // rows from before IDs existed
    if (k >= 0) taken[k] = true;
    const r = k >= 0 ? rows[k] : null;
    return [name, r ? r[1] === true : true, r ? r[2] : '', id];
  });
  const same = out.length === rows.length && out.every((r, i) =>
    String(rows[i][0]) === r[0] && (rows[i][1] === true) === r[1] && String(rows[i][2]) === String(r[2]) && String(rows[i][3]) === r[3]);
  if (same) return;
  if (n > 0) { const rg = sh.getRange(TAB_HEADER_ROW + 1, 1, n, 4); rg.clearContent(); rg.clearDataValidations(); }
  if (out.length) {
    sh.getRange(TAB_HEADER_ROW + 1, 1, out.length, 1).setNumberFormat('@');
    sh.getRange(TAB_HEADER_ROW + 1, 4, out.length, 1).setNumberFormat('@').setFontColor('#999999');
    sh.getRange(TAB_HEADER_ROW + 1, 2, out.length, 1).insertCheckboxes();
    sh.getRange(TAB_HEADER_ROW + 1, 1, out.length, 4).setValues(out);
  }
}

function getSettings_() {
  const ss = SpreadsheetApp.getActive();
  let sh = ss.getSheetByName(SETTINGS_SHEET);
  if (!sh) { setup(); sh = ss.getSheetByName(SETTINGS_SHEET); }
  syncTabs_(sh);
  const map = {};
  sh.getRange(1, 1, TAB_HEADER_ROW - 1, 2).getValues().forEach(r => {
    if (typeof r[0] === 'string' && r[0].trim()) map[r[0].trim()] = r[1];
  });
  const last = sh.getLastRow();
  const tabs = last > TAB_HEADER_ROW
    ? sh.getRange(TAB_HEADER_ROW + 1, 1, last - TAB_HEADER_ROW, 4).getValues()
        .filter(r => String(r[0]) !== '')
        .map(r => ({ name: String(r[0]), on: r[1] === true, headerRow: parseInt(r[2], 10) || 0, id: String(r[3]) }))
    : [];
  const logos = last > TAB_HEADER_ROW
    ? sh.getRange(TAB_HEADER_ROW + 1, 5, last - TAB_HEADER_ROW, 2).getValues()
        .filter(r => String(r[0]).trim() && String(r[1]).trim())
        .map(r => ({ word: String(r[0]).trim(), site: String(r[1]).trim().replace(/^https?:\/\//, '').replace(/\/.*$/, '') }))
    : [];
  return {
    sheet: sh,
    logos: logos,
    access: map[KEYS.access] === true,
    canEdit: map[KEYS.canEdit] === true,
    canDelete: map[KEYS.canDelete] === true,
    partnerPin: String(map[KEYS.partnerPin] || '').trim(),
    ownerPin: String(map[KEYS.ownerPin] || '').trim(),
    partnerName: String(map[KEYS.partnerName] || map['Who is using it (partner name)'] || 'Partner').trim(),
    ownerName: String(map[KEYS.ownerName] || 'Me').trim(),
    appName: rememberName_(String(map[KEYS.appName] || 'PLO')),
    tabs: tabs
  };
}

/** Keeps the app name handy so the page can open without reading the sheet. */
function rememberName_(name) {
  try {
    const pr = PropertiesService.getScriptProperties();
    if (pr.getProperty('appName') !== name) pr.setProperty('appName', name);
  } catch (e) {}
  return name;
}

function findKeyRow_(sh, label) {
  const vals = sh.getRange(1, 1, TAB_HEADER_ROW - 1, 1).getValues();
  for (let i = 0; i < vals.length; i++) if (String(vals[i][0]).trim() === label) return i + 1;
  // Older setups used a different label for the partner name
  if (label === KEYS.partnerName) {
    for (let i = 0; i < vals.length; i++) if (String(vals[i][0]).trim() === 'Who is using it (partner name)') return i + 1;
  }
  return 0;
}

function admin_(ctx) {
  const s = ctx.s;
  return {
    partnerName: s.partnerName, ownerName: s.ownerName, appName: s.appName,
    logos: s.logos,
    tabs: s.tabs.map(t => ({ name: t.name, on: t.on, id: t.id })),
    sheetUrl: SpreadsheetApp.getActive().getUrl() + '#gid=' + s.sheet.getSheetId()
  };
}

function saveAdmin_(ctx, p) {
  const sh = ctx.s.sheet;
  const set = (label, v) => {
    let r = findKeyRow_(sh, label);
    if (!r) { // add the row if this settings tab was made by an older version
      for (let i = 4; i < TAB_HEADER_ROW - 1; i++) if (String(sh.getRange(i, 1).getValue()) === '') { r = i; sh.getRange(i, 1).setValue(label).setFontWeight('bold'); break; }
    }
    if (r) sh.getRange(r, 2).setValue(v);
  };
  if ('partnerName' in p && String(p.partnerName).trim()) set(KEYS.partnerName, String(p.partnerName).trim().slice(0, 30));
  if ('ownerName' in p && String(p.ownerName).trim()) set(KEYS.ownerName, String(p.ownerName).trim().slice(0, 30));
  if ('appName' in p && String(p.appName).trim()) set(KEYS.appName, String(p.appName).trim().slice(0, 30));
  if (p.logos) {
    const lr = Math.max(sh.getLastRow(), TAB_HEADER_ROW + 1);
    sh.getRange(TAB_HEADER_ROW + 1, 5, lr - TAB_HEADER_ROW, 2).clearContent();
    const rows = p.logos.filter(l => l && String(l.word).trim() && String(l.site).trim())
      .map(l => [String(l.word).trim(), String(l.site).trim().replace(/^https?:\/\//, '').replace(/\/.*$/, '')]);
    if (rows.length) sh.getRange(TAB_HEADER_ROW + 1, 5, rows.length, 2).setValues(rows);
  }
  if (p.tabs) {
    const last = sh.getLastRow();
    if (last > TAB_HEADER_ROW) {
      const names = sh.getRange(TAB_HEADER_ROW + 1, 1, last - TAB_HEADER_ROW, 1).getValues();
      names.forEach((r, i) => {
        const n = String(r[0]);
        if (Object.prototype.hasOwnProperty.call(p.tabs, n)) sh.getRange(TAB_HEADER_ROW + 1 + i, 2).setValue(!!p.tabs[n]);
      });
    }
  }
  SpreadsheetApp.flush();
  return admin_(auth_());
}


function tabFor_(ctx, name) {
  const t = ctx.s.tabs.filter(x => x.name === name)[0];
  if (!t || !t.on || name === SETTINGS_SHEET) throw new Error('That tab is switched off in App Settings');
  const sh = SpreadsheetApp.getActive().getSheetByName(name);
  if (!sh) throw new Error('Tab not found: ' + name);
  return { sh: sh, headerRow: t.headerRow };
}

/* =============================== READ =============================== */

function bootstrap_(ctx, p) {
  const quick = !!(p && p.quick);
  autoWeekendFix_(ctx);
  const s = ctx.s, ss = SpreadsheetApp.getActive(), tz = ss.getSpreadsheetTimeZone();
  const tabs = [];
  s.tabs.forEach(t => {
    if (!t.on) return;
    const sh = ss.getSheetByName(t.name);
    if (!sh) return;
    try { tabs.push(readTab_(sh, t.headerRow, quick)); }
    catch (e) { tabs.push({ name: t.name, kind: 'error', error: String(e.message || e) }); }
  });
  const out = {
    role: ctx.role,
    appName: s.appName,
    partnerName: s.partnerName,
    ownerName: s.ownerName,
    canEdit: ctx.role === 'owner' || s.canEdit,
    canDelete: ctx.role === 'owner' || s.canDelete,
    access: s.access,
    today: Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd'),
    logos: s.logos,
    quick: quick,
    repeats: repeatsOut_(ctx),
    tabs: tabs
  };
  if (quick) cachePut_('quick', out);
  return out;
}

function headerInfo_(sh, override) {
  const lastRow = sh.getLastRow(), lastCol = sh.getLastColumn();
  if (lastRow < 1 || lastCol < 1) return null;
  const scan = sh.getRange(1, 1, Math.min(lastRow, 15), lastCol).getDisplayValues();
  let headerRow = override || 0;
  if (!headerRow) {
    headerRow = 1;
    for (let i = 0; i < scan.length; i++) {
      if (scan[i].filter(v => String(v).trim() !== '').length >= Math.min(2, lastCol)) { headerRow = i + 1; break; }
    }
  }
  const hv = headerRow <= scan.length ? scan[headerRow - 1] : sh.getRange(headerRow, 1, 1, lastCol).getDisplayValues()[0];
  let width = 0;
  hv.forEach((h, i) => { if (String(h).trim()) width = i + 1; });
  if (!width) width = lastCol;
  return { headerRow, width, lastRow, headers: hv.slice(0, width).map(h => String(h).trim()) };
}

function readTab_(sh, override, quick) {
  const hi = headerInfo_(sh, override);
  if (!hi) return { name: sh.getName(), id: sh.getSheetId(), kind: 'list', columns: [], rows: [] };
  const L = ledgerLayout_(sh, hi);
  return L ? readLedger_(sh, hi, L, quick) : readList_(sh, hi);
}

/* ----------------------------- LEDGER TABS ----------------------------- */
/* A ledger tab = one row per day with Day, Date and Running Total columns. */

function ledgerLayout_(sh, hi) {
  const h = hi.headers.map(x => x.toLowerCase());
  const day = h.findIndex(x => /^day$/.test(x));
  const date = h.findIndex(x => /^date/.test(x));
  // Running Total: column C on your tabs (falls back to any "Running Total" header).
  const run = /running\s*total/.test(h[2] || '') ? 2 : h.findIndex(x => /running\s*total/.test(x));
  if (date === -1 || run === -1) return null;

  // Read sign of each column from the Running Total formula on a data row.
  let signs = {};
  const probeRow = Math.min(hi.headerRow + 3, hi.lastRow);
  if (probeRow > hi.headerRow) {
    const f = sh.getRange(probeRow, run + 1).getFormula();
    if (f) signs = columnSigns_(f);
  }

  const notes = [], amounts = [];
  hi.headers.forEach((name, i) => {
    if (!name || i === day || i === date || i === run) return;
    if (/note|^from$|comment|detail/i.test(name)) { notes.push({ i: i, name: name }); return; }
    const sign = signs[i] || 0;
    amounts.push({ i: i, name: name, kind: sign > 0 ? 'in' : sign < 0 ? 'out' : 'info' });
  });

  // Paul's rule: the last money column before the notes is income into the account.
  if (notes.length) {
    const lastBefore = amounts.filter(a => a.i < notes[0].i).pop();
    if (lastBefore) lastBefore.kind = 'in';
  }

  // Which note column each money column writes to.
  const billNote = notes.filter(n => /bill/i.test(n.name))[0];
  const mainNote = notes.filter(n => !/bill/i.test(n.name))[0] || notes[0];
  amounts.forEach(a => {
    let n = mainNote;
    if (billNote && /bill|rent|loan|repay|petrol|grocer/i.test(a.name)) n = billNote;
    a.note = n ? n.i : -1;
  });
  return { day, date, run, notes, amounts };
}

/** Parses e.g. =C2-(D3:D426)-(G3:G426)+(H3:H426) → {3:-1, 6:-1, 7:+1}. */
function columnSigns_(f) {
  const out = {}, stack = [1];
  let op = 1, i = 0;
  f = String(f).replace(/^=/, '');
  while (i < f.length) {
    const c = f[i];
    if (c === '+') { op = 1; i++; continue; }
    if (c === '-') { op = -1; i++; continue; }
    if (c === '(') { stack.push(stack[stack.length - 1] * op); op = 1; i++; continue; }
    if (c === ')') { stack.pop(); if (!stack.length) stack.push(1); op = 1; i++; continue; }
    const m = /^\$?([A-Za-z]{1,3})\$?\d+(?::\$?[A-Za-z]{1,3}\$?\d+)?/.exec(f.slice(i));
    if (m) {
      const col = colIndex_(m[1].toUpperCase());
      const sgn = stack[stack.length - 1] * op;
      if (out[col] === undefined) out[col] = sgn;
      op = 1; i += m[0].length; continue;
    }
    i++;
  }
  return out;
}

function readLedger_(sh, hi, L, quick) {
  const tz = SpreadsheetApp.getActive().getSpreadsheetTimeZone();
  const today = Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd');
  const first = hi.headerRow + 1;
  const n = hi.lastRow - hi.headerRow;
  if (n <= 0) return { name: sh.getName(), id: sh.getSheetId(), kind: 'ledger', layout: L, headers: hi.headers, days: [], today: today };

  const dateVals = sh.getRange(first, L.date + 1, n, 1).getValues();
  let todayIdx = -1;
  for (let i = 0; i < n; i++) {
    const d = dateVals[i][0];
    if (d instanceof Date && Utilities.formatDate(d, tz, 'yyyy-MM-dd') <= today) todayIdx = i;
  }
  if (todayIdx === -1) todayIdx = 0;
  const startIdx = Math.max(0, todayIdx - (quick ? QUICK_BACK : DAYS_BACK));
  const endIdx = Math.min(n - 1, todayIdx + (quick ? QUICK_AHEAD : DAYS_AHEAD));
  const cnt = endIdx - startIdx + 1;

  const rg = sh.getRange(first + startIdx, 1, cnt, hi.width);
  const disp = rg.getDisplayValues(), vals = rg.getValues(), forms = rg.getFormulas();
  const days = [];
  for (let k = 0; k < cnt; k++) {
    const dv = dateVals[startIdx + k][0];
    if (!(dv instanceof Date)) continue;
    const raw = vals[k].map((v, c) => forms[k][c] ? forms[k][c] : (v instanceof Date ? '' : String(v)));
    days.push({ r: first + startIdx + k, date: Utilities.formatDate(dv, tz, 'yyyy-MM-dd'), d: disp[k], raw: raw });
  }
  const cf = cfRules_(sh);
  const fmtRow = first + todayIdx;
  const fmts = sh.getRange(fmtRow, 1, 1, hi.width).getNumberFormats()[0];
  return {
    name: sh.getName(), id: sh.getSheetId(), kind: 'ledger', headers: hi.headers, layout: L,
    moneyFmt: fmts[L.run] || '', today: today, days: days, cf: cf,
    partial: startIdx > 0 || endIdx < n - 1
  };
}

/** Reads the tab's conditional-formatting colour rules (number rules only) for the app. */
function cfRules_(sh) {
  const out = [];
  let theme = null;
  try { theme = SpreadsheetApp.getActive().getSpreadsheetTheme(); } catch (e) {}
  const hex = (colorObj, plain) => {
    try {
      if (colorObj) {
        const t = colorObj.getColorType();
        if (t === SpreadsheetApp.ColorType.RGB) return colorObj.asRgbColor().asHexString();
        if (t === SpreadsheetApp.ColorType.THEME && theme) return theme.getConcreteColor(colorObj.asThemeColor().getThemeColorType()).asRgbColor().asHexString();
      }
    } catch (e) {}
    return plain || '';
  };
  try {
    sh.getConditionalFormatRules().forEach(rule => {
      const bc = rule.getBooleanCondition();
      if (!bc) return;
      const type = String(bc.getCriteriaType());
      if (type.indexOf('NUMBER') !== 0) return;
      const cols = [];
      rule.getRanges().forEach(r => { for (let c = r.getColumn(); c < r.getColumn() + r.getNumColumns(); c++) cols.push(c - 1); });
      out.push({
        cols: cols, t: type, v: bc.getCriteriaValues().map(String),
        bg: hex(bc.getBackgroundObject ? bc.getBackgroundObject() : null, bc.getBackground()),
        fg: hex(bc.getFontColorObject ? bc.getFontColorObject() : null, bc.getFontColor()),
        b: bc.getBold() === true
      });
    });
  } catch (e) {}
  return out;
}

/* Amount as you'd write it in a note: 110, 20.33, 15.60 */
function noteAmt_(n) {
  n = Math.round(n * 100) / 100;
  return n % 1 === 0 ? String(n) : n.toFixed(2);
}

function escRe_(s) { return s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'); }

function addSpend_(ctx, p) {
  const t = tabFor_(ctx, p.tab);
  const amount = Math.round(parseFloat(p.amount) * 100) / 100;
  if (!(amount > 0)) throw new Error('Enter an amount');
  if (p.name && !p.keepName) p.name = tidyName_(p.name);
  const parts = (p.terms && p.terms.length > 1) ? p.terms.map(x => Math.round(parseFloat(x) * 100) / 100).filter(x => x > 0) : [amount];
  const lock = LockService.getDocumentLock();
  lock.waitLock(20000);
  try {
    const sh = t.sh, hi = headerInfo_(sh, t.headerRow), L = ledgerLayout_(sh, hi);
    if (!L) throw new Error('This tab is not a daily ledger');
    const col = L.amounts.filter(a => a.i === Number(p.col))[0];
    if (!col) throw new Error('Pick a column');
    const tz = SpreadsheetApp.getActive().getSpreadsheetTimeZone();
    const row = findDateRow_(sh, hi, L, p.date, tz);

    // 1) Money cell: blank → 45.26 ; 70 → =SUM(70+45.26) ; =SUM(70+10) → =SUM(70+10+45.26)
    const cell = sh.getRange(row, col.i + 1);
    const f = cell.getFormula(), a = parts.map(noteAmt_).join('+');
    const pm = parseMoney_(f, cell.getValue());
    if (pm) { if (!pm.style && pm.terms.length + parts.length > 1) pm.style = colStyle_(sh, hi, col.i); parts.forEach(x => pm.terms.push(x)); writeTerms_(cell, pm); }
    else if (f) cell.setFormula('=SUM((' + f.slice(1) + ')+' + a + ')');   // unusual formula: wrap it safely
    else cell.setValue(amount);
    // Keep the money format the column already uses.
    const nf = cell.getNumberFormat();
    if (!nf || nf === 'General' || nf === '0.###############') {
      const above = sh.getRange(Math.max(hi.headerRow + 1, row - 1), col.i + 1).getNumberFormat();
      cell.setNumberFormat(above && above !== 'General' ? above : '"$"#,##0.00');
    }

    // 2) Note: "Existing / Name (45.26)", or merge "Fitstop (54)" → "Fitstop (54+54)"
    const name = String(p.name || '').trim() || col.name.trim();
    if (col.note !== -1) {
      const nc = sh.getRange(row, col.note + 1);
      let note = String(nc.getValue() || '').replace(/\s+$/, '');
      const re = new RegExp('(^|/\\s*)(' + escRe_(name) + ')\\s*\\(([^)]*)\\)', 'i');
      if (re.test(note)) note = note.replace(re, (m, pre, nm, inner) => pre + nm + ' (' + inner + '+' + a + ')');
      else note = note ? note + ' / ' + name + ' (' + a + ')' : name + ' (' + a + ')';
      nc.setValue(note);
    }
    SpreadsheetApp.flush();
    return dayOut_(sh, hi, row, tz);
  } finally {
    lock.releaseLock();
  }
}

/** Edit a whole day in raw form: money cells as "70+45.26" and the note text. */
function saveDay_(ctx, p) {
  requireEdit_(ctx);
  const t = tabFor_(ctx, p.tab);
  const lock = LockService.getDocumentLock();
  lock.waitLock(20000);
  try {
    const sh = t.sh, hi = headerInfo_(sh, t.headerRow), L = ledgerLayout_(sh, hi);
    if (!L) throw new Error('This tab is not a daily ledger');
    const row = parseInt(p.row, 10);
    if (!(row > hi.headerRow)) throw new Error('Bad row');
    const tz = SpreadsheetApp.getActive().getSpreadsheetTimeZone();
    if (p.expect) {
      const now = dayOut_(sh, hi, row, tz).raw;
      if (now.join('\u0001') !== p.expect.join('\u0001')) throw new Error('CHANGED');
    }
    const cells = p.cells || {};
    L.amounts.forEach(a => {
      if (!Object.prototype.hasOwnProperty.call(cells, a.i)) return;
      const expr = String(cells[a.i]).replace(/[$,\s]/g, '');
      const c = sh.getRange(row, a.i + 1);
      if (expr === '') c.clearContent();
      else if (/^-?\d+(\.\d+)?$/.test(expr)) c.setValue(parseFloat(expr));
      else if (/^[0-9.+\-*/()]+$/.test(expr)) c.setFormula('=SUM(' + expr + ')');
      else throw new Error('"' + a.name + '" can only contain numbers and + - * /');
    });
    L.notes.forEach(nt => {
      if (Object.prototype.hasOwnProperty.call(cells, nt.i)) sh.getRange(row, nt.i + 1).setValue(String(cells[nt.i]));
    });
    SpreadsheetApp.flush();
    return dayOut_(sh, hi, row, tz);
  } finally {
    lock.releaseLock();
  }
}

function splitNote_(note) {
  const out = []; let cur = '', d = 0; note = String(note || '');
  for (let i = 0; i < note.length; i++) {
    const c = note[i];
    if (c === '(') d++; if (c === ')') d = Math.max(0, d - 1);
    if (c === '/' && d === 0) { if (cur.trim()) out.push(cur.trim()); cur = ''; } else cur += c;
  }
  if (cur.trim()) out.push(cur.trim());
  return out;
}
function evalSimple_(e) {
  e = String(e).replace(/\s/g, '').replace(/[+\-]+$/, '');
  if (!/^[0-9.+\-]+$/.test(e)) return null;
  let tot = 0; const m = e.match(/[+\-]?[0-9.]+/g); if (!m) return null;
  m.forEach(x => tot += parseFloat(x));
  return Math.round(tot * 100) / 100;
}
/** Reads a money cell the way you write them: 70 | =sum(160+110) | =SUM(..) | =(151.74+222.96) | =54+54
 *  Returns {style, terms} so it can be written back in exactly the same style, or null if it's something else. */
function parseMoney_(f, v) {
  if (f) {
    const t = String(f).trim();
    let m = /^=\s*(sum|Sum|SUM)?\s*\(\s*([0-9.+\s]+)\)\s*$/.exec(t);
    if (m) return { style: '=' + (m[1] || '') + '(', terms: m[2].split('+').map(x => parseFloat(x)).filter(x => !isNaN(x)) };
    m = /^=\s*([0-9.+\s]+)$/.exec(t);
    if (m) return { style: '=', terms: m[1].split('+').map(x => parseFloat(x)).filter(x => !isNaN(x)) };
    return null;
  }
  if (v === '' || v === null) return { style: null, terms: [] };
  const n = parseFloat(String(v).replace(/[$,]/g, ''));
  return isNaN(n) ? null : { style: null, terms: [n] };
}
function buildMoney_(style, terms) {
  if (!terms.length) return '';
  const body = terms.map(noteAmt_).join('+');
  if (style === '=') return '=' + body;
  if (style) return style + body + ')';
  return terms.length === 1 ? terms[0] : '=SUM(' + body + ')';
}
function cellTerms_(cell) { return parseMoney_(cell.getFormula(), cell.getValue()); }
function writeTerms_(cell, pm) {
  const out = buildMoney_(pm.style, pm.terms);
  if (out === '') cell.clearContent();
  else if (typeof out === 'number') cell.setValue(out);
  else cell.setFormula(out);
}
function editItems_(ctx, p) {
  requireEdit_(ctx);
  const t = tabFor_(ctx, p.tab);
  const lock = LockService.getDocumentLock();
  lock.waitLock(20000);
  try {
    const sh = t.sh, hi = headerInfo_(sh, t.headerRow), L = ledgerLayout_(sh, hi);
    if (!L) throw new Error('This tab is not a daily ledger');
    let done = 0; const found = [];
    (p.items || []).forEach(it => {
      const row = parseInt(it.row, 10), noteCol = parseInt(it.note, 10);
      if (!(row > hi.headerRow)) throw new Error('Bad row');
      const nc = sh.getRange(row, noteCol + 1);
      const parts = splitNote_(nc.getValue());
      const oldAmt = Math.round(parseFloat(it.oldAmt) * 100) / 100;
      const idx = parts.findIndex(pt => {
        const m = /^(.*?)\s*\(([^()]*)\)\s*$/.exec(pt);
        return m && m[1].trim().toLowerCase() === String(it.name).trim().toLowerCase() && Math.abs((evalSimple_(m[2]) || 0) - oldAmt) < 0.005;
      });
      if (idx < 0) throw new Error('Couldn\u2019t find "' + it.name + ' (' + noteAmt_(oldAmt) + ')" in that day\u2019s note. It may have changed in the sheet.');
      const itemExpr = /\(([^()]*)\)\s*$/.exec(parts[idx])[1];
      const itemTerms = itemExpr.split('+').map(x => parseFloat(x)).filter(x => !isNaN(x));
      // Which money column holds it? Look for the amount (or its parts, e.g. 54+54) in the mapped columns.
      const cols = L.amounts.filter(a => a.note === noteCol);
      let target = null, terms = null, at = [], pm = null;
      for (const a of cols) {
        const got = cellTerms_(sh.getRange(row, a.i + 1)); if (!got) continue;
        const tt = got.terms; pm = got;
        const k = tt.findIndex(x => Math.abs(x - oldAmt) < 0.005);
        if (k >= 0) { target = a; terms = tt; at = [k]; break; }
        const used = []; let ok = itemTerms.length > 1;
        itemTerms.forEach(x => { const j = tt.findIndex((y, q) => used.indexOf(q) < 0 && Math.abs(y - x) < 0.005); if (j < 0) ok = false; else used.push(j); });
        if (ok) { target = a; terms = tt; at = used; break; }
      }
      if (!target) throw new Error('Couldn\u2019t tell which column holds ' + it.name + '. Tap that day and edit it directly.');
      const remove = it.remove === true || !(parseFloat(it.newAmt) > 0);
      const newAmt = remove ? 0 : Math.round(parseFloat(it.newAmt) * 100) / 100;
      at.sort((a, b) => b - a).forEach((q, n) => { if (n === at.length - 1 && !remove) terms[q] = newAmt; else terms.splice(q, 1); });
      pm.terms = terms;
      found.push({ col: target.i, terms: itemTerms.length ? itemTerms : [oldAmt] });
      writeTerms_(sh.getRange(row, target.i + 1), pm);
      if (remove) parts.splice(idx, 1);
      else parts[idx] = (String(it.newName || it.name).trim()) + ' (' + noteAmt_(newAmt) + ')';
      nc.setValue(parts.join(' / '));
      done++;
    });
    SpreadsheetApp.flush();
    return { done: done, found: found };
  } finally {
    lock.releaseLock();
  }
}

function findDateRow_(sh, hi, L, ymd, tz) {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(String(ymd))) throw new Error('Pick a date');
  const n = hi.lastRow - hi.headerRow;
  const vals = sh.getRange(hi.headerRow + 1, L.date + 1, n, 1).getValues();
  for (let i = n - 1; i >= 0; i--) {
    const d = vals[i][0];
    if (d instanceof Date && Utilities.formatDate(d, tz, 'yyyy-MM-dd') === ymd) return hi.headerRow + 1 + i;
  }
  throw new Error('There is no row for that date in "' + sh.getName() + '" yet');
}

function dayOut_(sh, hi, row, tz) {
  const rg = sh.getRange(row, 1, 1, hi.width);
  const v = rg.getValues()[0], f = rg.getFormulas()[0];
  const L = ledgerLayout_(sh, hi);
  const dv = v[L.date];
  return {
    r: row,
    date: dv instanceof Date ? Utilities.formatDate(dv, tz, 'yyyy-MM-dd') : '',
    d: rg.getDisplayValues()[0],
    raw: v.map((x, c) => f[c] ? f[c] : (x instanceof Date ? '' : String(x)))
  };
}

/* ------------------------------ LIST TABS ------------------------------ */
/* Anything else (e.g. Houses): one row per item, add/edit like a form.     */

function listMeta_(sh, hi) {
  const first = hi.headerRow + 1;
  const start = Math.max(first, hi.lastRow - MAX_LIST_ROWS + 1);
  const n = hi.lastRow >= start ? hi.lastRow - start + 1 : 0;
  let values = [], display = [], formulas = [];
  if (n > 0) {
    const rg = sh.getRange(start, 1, n, hi.width);
    values = rg.getValues(); display = rg.getDisplayValues(); formulas = rg.getFormulas();
  }
  let refIdx = -1;
  for (let i = n - 1; i >= 0; i--) {
    if (display[i].some((v, c) => v !== '' && !formulas[i][c])) { refIdx = i; break; }
  }
  const refRow = refIdx >= 0 ? start + refIdx : first;
  const ref = sh.getRange(refRow, 1, 1, hi.width);
  const fmts = ref.getNumberFormats()[0], dvs = ref.getDataValidations()[0];
  const rf = refIdx >= 0 ? formulas[refIdx] : ref.getFormulas()[0];
  const rv = refIdx >= 0 ? values[refIdx] : ref.getValues()[0];
  const columns = hi.headers.map((h, c) => describeColumn_(h || ('Column ' + colLetter_(c + 1)), c, fmts[c], dvs[c], rf[c], rv[c]));
  let lastDataRow = hi.headerRow;
  for (let i = n - 1; i >= 0; i--) {
    if (columns.some((col, c) => col.type !== 'formula' && display[i][c] !== '')) { lastDataRow = start + i; break; }
  }
  return { start, n, values, display, formulas, columns, refRow, lastDataRow };
}

function describeColumn_(header, c, fmt, dv, formula, sample) {
  const col = { i: c, name: header, type: 'text', fmt: fmt || '' };
  const h = header.toLowerCase();
  if (formula) col.type = 'formula';
  else if (dv) {
    const T = SpreadsheetApp.DataValidationCriteria, crit = dv.getCriteriaType(), args = dv.getCriteriaValues();
    if (crit === T.CHECKBOX) col.type = 'checkbox';
    else if (crit === T.VALUE_IN_LIST) { col.type = 'select'; col.options = (args[0] || []).map(String); }
    else if (crit === T.VALUE_IN_RANGE) {
      col.type = 'select';
      try { col.options = args[0].getDisplayValues().reduce((a, r) => a.concat(r), []).map(String).filter((v, i, a) => v && a.indexOf(v) === i); }
      catch (e) { col.options = []; }
    }
    else if (String(crit).indexOf('DATE') === 0) col.type = 'date';
  }
  if (col.type === 'text') {
    const f = (fmt || '').toLowerCase();
    if (sample instanceof Date || /(d-m|d\/m|m\/d|mmm|yy)/.test(f)) col.type = 'date';
    else if (typeof sample === 'boolean') col.type = 'checkbox';
    else if (/\$|€|£|đ|rp/.test(f)) col.type = 'currency';
    else if (f.indexOf('%') !== -1) col.type = 'percent';
    else if (typeof sample === 'number') col.type = /price|amount|cost|total|\$/.test(h) ? 'currency' : 'number';
    else if (sample === '' && /date|when/.test(h)) col.type = 'date';
    else if (sample === '' && /price|amount|cost|\$/.test(h)) col.type = 'currency';
  }
  return col;
}

function readList_(sh, hi) {
  const tz = SpreadsheetApp.getActive().getSpreadsheetTimeZone();
  const m = listMeta_(sh, hi);
  const rows = [];
  for (let i = 0; i < m.n; i++) {
    const r = m.start + i;
    if (r > m.lastDataRow) break;
    if (!m.columns.some((col, c) => col.type !== 'formula' && m.display[i][c] !== '')) continue;
    rows.push({ r: r, d: m.display[i], v: m.values[i].map(x => toClient_(x, tz)) });
  }
  return { name: sh.getName(), id: sh.getSheetId(), kind: 'list', columns: m.columns, rows: rows, headerRow: hi.headerRow };
}

function toClient_(x, tz) {
  if (x instanceof Date) return Utilities.formatDate(x, tz, 'yyyy-MM-dd');
  return x === null || x === undefined ? '' : x;
}

function fromClient_(col, v, tz) {
  if (v === null || v === undefined) return '';
  if (col.type === 'checkbox') return v === true || v === 'true';
  if (v === '') return '';
  if (col.type === 'date' && /^\d{4}-\d{2}-\d{2}$/.test(String(v))) return Utilities.parseDate(v + ' 12:00', tz, 'yyyy-MM-dd HH:mm');
  if (col.type === 'currency' || col.type === 'number' || col.type === 'percent') {
    const n = parseFloat(String(v).replace(/[^0-9.\-]/g, ''));
    if (isNaN(n)) return String(v);
    return col.type === 'percent' && Math.abs(n) > 1 ? n / 100 : n;
  }
  return String(v);
}

function addRow_(ctx, tab, values) {
  const t = tabFor_(ctx, tab);
  const lock = LockService.getDocumentLock();
  lock.waitLock(20000);
  try {
    const sh = t.sh, hi = headerInfo_(sh, t.headerRow), m = listMeta_(sh, hi);
    const tz = SpreadsheetApp.getActive().getSpreadsheetTimeZone();
    const target = m.lastDataRow + 1;
    if (target <= sh.getMaxRows()) {
      const below = sh.getRange(target, 1, 1, hi.width).getDisplayValues()[0];
      if (m.columns.some((col, c) => col.type !== 'formula' && below[c] !== '')) sh.insertRowAfter(m.lastDataRow);
    } else sh.insertRowAfter(sh.getMaxRows());
    if (m.lastDataRow > hi.headerRow) {
      const src = sh.getRange(m.lastDataRow, 1, 1, hi.width), dst = sh.getRange(target, 1, 1, hi.width);
      src.copyTo(dst, SpreadsheetApp.CopyPasteType.PASTE_FORMAT, false);
      src.copyTo(dst, SpreadsheetApp.CopyPasteType.PASTE_DATA_VALIDATION, false);
      const f = src.getFormulasR1C1()[0];
      m.columns.forEach(col => { if (col.type === 'formula' && f[col.i]) sh.getRange(target, col.i + 1).setFormulaR1C1(f[col.i]); });
    }
    m.columns.forEach(col => {
      if (col.type === 'formula' || !values || !Object.prototype.hasOwnProperty.call(values, col.i)) return;
      sh.getRange(target, col.i + 1).setValue(fromClient_(col, values[col.i], tz));
    });
    SpreadsheetApp.flush();
    return rowOut_(sh, target, hi.width);
  } finally { lock.releaseLock(); }
}

function checkUnchanged_(sh, row, width, expect) {
  if (!expect) return;
  const now = sh.getRange(row, 1, 1, width).getDisplayValues()[0];
  if (now.join('\u0001') !== expect.join('\u0001')) throw new Error('CHANGED');
}

function updateRow_(ctx, tab, row, values, expect) {
  requireEdit_(ctx);
  const t = tabFor_(ctx, tab);
  const lock = LockService.getDocumentLock();
  lock.waitLock(20000);
  try {
    const sh = t.sh, hi = headerInfo_(sh, t.headerRow), m = listMeta_(sh, hi);
    const tz = SpreadsheetApp.getActive().getSpreadsheetTimeZone();
    row = parseInt(row, 10);
    if (!(row > hi.headerRow)) throw new Error('Bad row');
    checkUnchanged_(sh, row, hi.width, expect);
    m.columns.forEach(col => {
      if (col.type === 'formula' || !Object.prototype.hasOwnProperty.call(values || {}, col.i)) return;
      sh.getRange(row, col.i + 1).setValue(fromClient_(col, values[col.i], tz));
    });
    SpreadsheetApp.flush();
    return rowOut_(sh, row, hi.width);
  } finally { lock.releaseLock(); }
}

function deleteRow_(ctx, tab, row, expect) {
  if (ctx.role !== 'owner' && !ctx.s.canDelete) throw new Error('Deleting is switched off in App Settings');
  const t = tabFor_(ctx, tab);
  const lock = LockService.getDocumentLock();
  lock.waitLock(20000);
  try {
    const hi = headerInfo_(t.sh, t.headerRow);
    row = parseInt(row, 10);
    if (!(row > hi.headerRow)) throw new Error('Bad row');
    if (ledgerLayout_(t.sh, hi)) throw new Error('Days in a ledger tab can be edited, not deleted');
    checkUnchanged_(t.sh, row, hi.width, expect);
    t.sh.deleteRow(row);
    return { deleted: row };
  } finally { lock.releaseLock(); }
}

function rowOut_(sh, row, width) {
  const tz = SpreadsheetApp.getActive().getSpreadsheetTimeZone();
  const rg = sh.getRange(row, 1, 1, width);
  return { r: row, d: rg.getDisplayValues()[0], v: rg.getValues()[0].map(x => toClient_(x, tz)) };
}

function colLetter_(n) {
  let s = '';
  while (n > 0) { const m = (n - 1) % 26; s = String.fromCharCode(65 + m) + s; n = Math.floor((n - 1) / 26); }
  return s;
}

function colIndex_(letters) {
  let n = 0;
  for (let i = 0; i < letters.length; i++) n = n * 26 + (letters.charCodeAt(i) - 64);
  return n - 1;
}


/* ============================ PLANNING TOOLS ============================ */
/* Move a bill to another day, keep weekday-only bills off weekends, change a regular
   amount from a date, add a new sheet, rename sheets, and fill a new year with the
   regular bills. Everything is written in the same style you type by hand. */

const DAY_NAMES = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'];
function r2_(x) { return Math.round(parseFloat(x) * 100) / 100; }
function ymdOf_(d, tz) { return Utilities.formatDate(d, tz, 'yyyy-MM-dd'); }
function dt_(s) { const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(String(s)); if (!m) throw new Error('Pick a date'); return new Date(+m[1], +m[2] - 1, +m[3], 12); }
function ymd_(d) { return d.getFullYear() + '-' + ('0' + (d.getMonth() + 1)).slice(-2) + '-' + ('0' + d.getDate()).slice(-2); }
function plusDays_(s, n) { const d = dt_(s); d.setDate(d.getDate() + n); return ymd_(d); }
function daysApart_(a, b) { return Math.round((dt_(b) - dt_(a)) / 86400000); }
function dowOf_(s) { return dt_(s).getDay(); }
function colLetter2_(i) { let n = i + 1, s = ''; while (n > 0) { const m = (n - 1) % 26; s = String.fromCharCode(65 + m) + s; n = Math.floor((n - 1) / 26); } return s; }

function ledgerFor_(ctx, tabName) {
  const t = tabFor_(ctx, tabName);
  const sh = t.sh, hi = headerInfo_(sh, t.headerRow), L = ledgerLayout_(sh, hi);
  if (!L) throw new Error('"' + tabName + '" isn’t a daily tab');
  return { sh: sh, hi: hi, L: L, tz: SpreadsheetApp.getActive().getSpreadsheetTimeZone() };
}

/** How this column writes several amounts: =(a+b), =sum(a+b), =SUM(a+b) or =a+b. Uses whatever you use most. */
function colStyle_(sh, hi, ci) {
  const n = hi.lastRow - hi.headerRow; if (n < 1) return '=SUM(';
  const fs = sh.getRange(hi.headerRow + 1, ci + 1, n, 1).getFormulas(), count = {};
  fs.forEach(r => { const pm = r[0] ? parseMoney_(r[0], '') : null; if (pm && pm.style) count[pm.style] = (count[pm.style] || 0) + 1; });
  let best = '=SUM(', bn = 0; Object.keys(count).forEach(k => { if (count[k] > bn) { bn = count[k]; best = k; } });
  return best;
}

/** Splits one row into items: named ones from the note ("Car Loan (110)") matched to their column, and plain amounts with no note (like a wage). */
function rowItems_(L, pms, notes) {
  const rem = {}, items = [];
  L.amounts.forEach(a => { rem[a.i] = pms[a.i] ? pms[a.i].terms.slice() : []; });
  L.notes.forEach(nt => {
    splitNote_(notes[nt.i]).forEach(pt => {
      const m = /^(.*?)\s*\(([^()]*)\)\s*$/.exec(pt); if (!m) return;
      const name = m[1].trim(), terms = m[2].split('+').map(x => parseFloat(x)).filter(x => !isNaN(x) && x > 0);
      if (!name || !terms.length) return;
      const amt = r2_(terms.reduce((a, b) => a + b, 0));
      const cols = L.amounts.filter(a => a.note === nt.i);
      for (const a of cols) {
        const r = rem[a.i];
        const k = r.findIndex(x => Math.abs(x - amt) < 0.005);
        if (k >= 0) { r.splice(k, 1); items.push({ name: name, amt: amt, terms: terms, col: a.i }); return; }
        const used = []; let ok = terms.length > 1;
        terms.forEach(x => { const j = r.findIndex((y, q) => used.indexOf(q) < 0 && Math.abs(y - x) < 0.005); if (j < 0) ok = false; else used.push(j); });
        if (ok) { used.sort((a2, b2) => b2 - a2).forEach(q => r.splice(q, 1)); items.push({ name: name, amt: amt, terms: terms, col: a.i }); return; }
      }
    });
  });
  L.amounts.forEach(a => rem[a.i].forEach(x => { if (x > 0) items.push({ name: '', amt: r2_(x), terms: [r2_(x)], col: a.i }); }));
  return items;
}

/** Reads a block of day rows: dates, money cells and notes. */
function readBlock_(X, fromRow, n) {
  const sh = X.sh, L = X.L, width = X.hi.width || sh.getLastColumn();
  const rg = sh.getRange(fromRow, 1, n, width), vals = rg.getValues(), fms = rg.getFormulas();
  return vals.map((v, i) => {
    const d = v[L.date];
    const pms = {}, notes = {};
    L.amounts.forEach(a => { pms[a.i] = parseMoney_(fms[i][a.i], v[a.i]); });
    L.notes.forEach(nt => { notes[nt.i] = String(v[nt.i] || ''); });
    return { r: fromRow + i, date: d instanceof Date ? ymdOf_(d, X.tz) : '', pms: pms, notes: notes, v: v, f: fms[i] };
  });
}

/* ---------------- move a bill to another day ---------------- */
function moveItem_(ctx, p) {
  const amt = r2_(p.amt), to = String(p.date || '');
  if (!/^\d{4}-\d{2}-\d{2}$/.test(to)) throw new Error('Pick the new date');
  const X = ledgerFor_(ctx, p.tab);
  const fromDate = ymdOf_(X.sh.getRange(parseInt(p.row, 10), X.L.date + 1).getValue(), X.tz);
  findDateRow_(X.sh, X.hi, X.L, to, X.tz);            // make sure the new day exists before touching anything
  const newAmt = p.newAmt != null && p.newAmt !== '' ? r2_(p.newAmt) : amt;
  const name = String(p.newName || p.name).trim() || p.name;
  const r = editItems_(ctx, { tab: p.tab, items: [{ row: p.row, note: p.note, name: p.name, oldAmt: amt, remove: true }] });
  const f = r.found[0];
  const terms = newAmt === amt ? f.terms : [newAmt];
  try {
    addSpend_(ctx, { tab: p.tab, date: to, col: f.col, amount: newAmt, terms: terms, name: name, keepName: true });
  } catch (e) {
    try { addSpend_(ctx, { tab: p.tab, date: fromDate, col: f.col, amount: amt, terms: f.terms, name: p.name, keepName: true }); } catch (e2) {}
    throw e;
  }
  return { ok: true, from: fromDate, to: to };
}

/* ---------------- weekday-only bills ---------------- */
function weekdayCfg_() {
  try { const c = JSON.parse(PropertiesService.getScriptProperties().getProperty('weekdayBills') || '{}'); c.names = c.names || []; c.dir = c.dir || 'after'; return c; }
  catch (e) { return { names: [], dir: 'after' }; }
}
function weekdaySave_(ctx, p) {
  const cfg = { names: (p.names || []).map(x => String(x).trim()).filter(Boolean).slice(0, 300), dir: p.dir === 'before' ? 'before' : 'after' };
  const pr = PropertiesService.getScriptProperties();
  pr.setProperty('weekdayBills', JSON.stringify(cfg));
  const r = weekendFix_(ctx, {});
  pr.setProperty('weekdayFixed', ymdOf_(new Date(), SpreadsheetApp.getActive().getSpreadsheetTimeZone()));
  return { cfg: cfg, moved: r.moved, failed: r.failed };
}
function shiftWeekday_(ds, dir, today) {
  const w = dowOf_(ds); if (w !== 0 && w !== 6) return ds;
  if (dir === 'before') { const b = plusDays_(ds, w === 6 ? -1 : -2); if (!today || b >= today) return b; }
  return plusDays_(ds, w === 6 ? 2 : 1);
}
/** Finds weekday-only bills that land on a Saturday or Sunday from today on, and moves them to the Monday after (or Friday before). */
function weekendFix_(ctx, p) {
  const cfg = weekdayCfg_(), names = cfg.names.map(x => x.toLowerCase());
  const list = [], moved = [], failed = [];
  if (!names.length) return { moved: [], failed: [] };
  ctx.s.tabs.filter(t => t.on).forEach(t => {
    let X; try { X = ledgerFor_(ctx, t.name); } catch (e) { return; }
    const today = ymdOf_(new Date(), X.tz), first = X.hi.headerRow + 1, n = X.hi.lastRow - X.hi.headerRow;
    if (n < 1) return;
    const dates = X.sh.getRange(first, X.L.date + 1, n, 1).getValues();
    X.L.notes.forEach(nt => {
      const notes = X.sh.getRange(first, nt.i + 1, n, 1).getValues();
      for (let i = 0; i < n; i++) {
        const d = dates[i][0]; if (!(d instanceof Date)) continue;
        const ds = ymdOf_(d, X.tz); if (ds < today) continue;
        const w = dowOf_(ds); if (w !== 0 && w !== 6) continue;
        splitNote_(notes[i][0]).forEach(pt => {
          const m = /^(.*?)\s*\(([^()]*)\)\s*$/.exec(pt); if (!m) return;
          const nm = m[1].trim(); if (names.indexOf(nm.toLowerCase()) < 0) return;
          const amt = evalSimple_(m[2]); if (!(amt > 0)) return;
          list.push({ tab: t.name, row: first + i, note: nt.i, name: nm, amt: amt, from: ds, date: shiftWeekday_(ds, cfg.dir, today) });
        });
      }
    });
  });
  if (p.dryRun) return { moved: list, failed: [] };
  list.forEach(x => { try { moveItem_(ctx, x); moved.push(x); } catch (e) { x.error = String(e.message || e); failed.push(x); } });
  return { moved: moved, failed: failed };
}
function autoWeekendFix_(ctx) {
  try {
    const pr = PropertiesService.getScriptProperties();
    const today = ymdOf_(new Date(), SpreadsheetApp.getActive().getSpreadsheetTimeZone());
    if (pr.getProperty('weekdayFixed') === today || !weekdayCfg_().names.length) return;
    pr.setProperty('weekdayFixed', today);
    weekendFix_(ctx, {});
  } catch (e) {}
}

/* ---------------- change a regular amount from a date (e.g. a pay rise) ---------------- */
function bulkChange_(ctx, p) {
  requireEdit_(ctx);
  const X = ledgerFor_(ctx, p.tab), L = X.L, sh = X.sh;
  const col = L.amounts.filter(a => a.i === Number(p.col))[0]; if (!col) throw new Error('Pick a column');
  const oldAmt = r2_(p.oldAmt), newAmt = p.remove ? 0 : r2_(p.newAmt);
  if (!(oldAmt > 0)) throw new Error('Enter the current amount');
  if (!p.remove && !(newAmt > 0)) throw new Error('Enter the new amount');
  const from = String(p.from || ''), to = String(p.to || '') || '9999-12-31';
  if (!/^\d{4}-\d{2}-\d{2}$/.test(from)) throw new Error('Pick the date it starts');
  const nameF = String(p.name || '').trim().toLowerCase(), newName = String(p.newName || '').trim();
  const lock = LockService.getDocumentLock(); lock.waitLock(20000);
  try {
    const first = X.hi.headerRow + 1, n = X.hi.lastRow - X.hi.headerRow;
    const rows = readBlock_(X, first, n), hits = [];
    rows.forEach(rw => {
      if (!rw.date || rw.date < from || rw.date > to) return;
      const pm = rw.pms[col.i]; if (!pm) return;
      const k = pm.terms.findIndex(x => Math.abs(x - oldAmt) < 0.005); if (k < 0) return;
      let parts = null, pi = -1;
      if (col.note !== -1) {
        parts = splitNote_(rw.notes[col.note]);
        pi = parts.findIndex(pt => { const m = /^(.*?)\s*\(([^()]*)\)\s*$/.exec(pt);
          return m && Math.abs((evalSimple_(m[2]) || 0) - oldAmt) < 0.005 && (!nameF || m[1].trim().toLowerCase() === nameF); });
      }
      if (nameF && pi < 0) return;      // a name was given: only change days where that name is in the note
      hits.push({ rw: rw, pm: pm, k: k, parts: parts, pi: pi });
    });
    if (p.dryRun) return { count: hits.length, dates: hits.map(h => h.rw.date) };
    hits.forEach(h => {
      if (p.remove) h.pm.terms.splice(h.k, 1); else h.pm.terms[h.k] = newAmt;
      writeTerms_(sh.getRange(h.rw.r, col.i + 1), h.pm);
      if (h.pi >= 0) {
        const m = /^(.*?)\s*\(([^()]*)\)\s*$/.exec(h.parts[h.pi]);
        if (p.remove) h.parts.splice(h.pi, 1); else h.parts[h.pi] = (newName || m[1].trim()) + ' (' + noteAmt_(newAmt) + ')';
        sh.getRange(h.rw.r, col.note + 1).setValue(h.parts.join(' / '));
      }
    });
    SpreadsheetApp.flush();
    return { count: hits.length, dates: hits.map(h => h.rw.date) };
  } finally { lock.releaseLock(); }
}

/* ---------------- more day rows (used by new sheets and new years) ---------------- */
function extendRows_(sh, hi, L, until, tz) {
  let last = sh.getLastRow();
  while (last > hi.headerRow && !(sh.getRange(last, L.date + 1).getValue() instanceof Date)) last--;
  if (last <= hi.headerRow) throw new Error('No dated rows to continue from');
  const lastYmd = ymdOf_(sh.getRange(last, L.date + 1).getValue(), tz), need = daysApart_(lastYmd, until);
  if (need <= 0) return 0;
  const oldMax = sh.getMaxRows();
  if (oldMax < last + need) sh.insertRowsAfter(oldMax, last + need - oldMax);
  const width = sh.getLastColumn();
  sh.getRange(last, 1, 1, width).copyTo(sh.getRange(last + 1, 1, need, width), SpreadsheetApp.CopyPasteType.PASTE_FORMAT, false);
  const dc = colLetter2_(L.date), dates = [], names = [];
  for (let i = 1; i <= need; i++) { dates.push(['=' + dc + (last + i - 1) + '+1']); names.push([DAY_NAMES[dowOf_(plusDays_(lastYmd, i))]]); }
  sh.getRange(last + 1, L.date + 1, need, 1).setFormulas(dates);
  if (L.day !== -1) sh.getRange(last + 1, L.day + 1, need, 1).setValues(names);
  sh.getRange(last, L.run + 1).copyTo(sh.getRange(last + 1, L.run + 1, need, 1));   // same Running Total formula, moved down
  // stretch the colour bands to cover the new rows
  const newMax = sh.getMaxRows();
  if (newMax > oldMax) {
    const rules = sh.getConditionalFormatRules().map(rule => {
      const rs = rule.getRanges().map(g => g.getLastRow() >= oldMax - 1 ? sh.getRange(g.getRow(), g.getColumn(), newMax - g.getRow() + 1, g.getNumColumns()) : g);
      return rule.copy().setRanges(rs).build();
    });
    sh.setConditionalFormatRules(rules);
  }
  return need;
}

/* ---------------- new sheet, same layout ---------------- */
function createTab_(ctx, p) {
  requireEdit_(ctx);
  const ss = SpreadsheetApp.getActive(), name = String(p.name || '').trim();
  if (!name) throw new Error('Give the new sheet a name');
  if (/[\[\]*?:\/\\]/.test(name) || name.length > 90) throw new Error('Sheet names can’t use [ ] * ? : / \\');
  if (ss.getSheetByName(name)) throw new Error('There’s already a sheet called "' + name + '"');
  const X = ledgerFor_(ctx, p.from);
  const start = String(p.start || ''); dt_(start);
  const years = Math.max(1, Math.min(5, parseInt(p.years, 10) || 2));
  const end = plusDays_(ymd_(new Date(dt_(start).getFullYear() + years, dt_(start).getMonth(), dt_(start).getDate(), 12)), -1);
  const lock = LockService.getDocumentLock(); lock.waitLock(20000);
  try {
    const sh = X.sh.copyTo(ss).setName(name);
    ss.setActiveSheet(sh); ss.moveActiveSheet(X.sh.getIndex() + 1);
    const hi = headerInfo_(sh, X.hi.headerRow), L = ledgerLayout_(sh, hi);
    const first = hi.headerRow + 1;
    let lastRow = sh.getLastRow();
    const n = Math.max(1, lastRow - first + 1);
    L.amounts.concat(L.notes).forEach(c => sh.getRange(first, c.i + 1, n, 1).clearContent());
    const lastUsed = Math.max.apply(null, [L.run, L.date].concat(L.amounts.map(a => a.i), L.notes.map(x => x.i)));
    if (sh.getMaxColumns() > lastUsed + 1) sh.getRange(1, lastUsed + 2, sh.getMaxRows(), sh.getMaxColumns() - lastUsed - 1).clear();
    if (!p.colours) sh.setConditionalFormatRules([]);
    // first day: the start date and opening balance; every other day follows from it
    sh.getRange(first, L.date + 1).setValue(dt_(start));
    sh.getRange(first, L.run + 1).setValue(r2_(p.opening || 0));
    const want = daysApart_(start, end) + 1;
    if (n > want) sh.getRange(first + want, 1, n - want, sh.getMaxColumns()).clearContent();
    SpreadsheetApp.flush();
    if (n < want) extendRows_(sh, hi, L, end, X.tz);
    if (L.day !== -1) {
      const names = []; for (let i = 0; i < want; i++) names.push([DAY_NAMES[dowOf_(plusDays_(start, i))]]);
      sh.getRange(first, L.day + 1, want, 1).setValues(names);
    }
    SpreadsheetApp.flush();
  } finally { lock.releaseLock(); }
  const s = getSettings_();                 // adds the new sheet to App Settings
  saveAdmin_({ role: 'owner', s: s }, { tabs: { [name]: true } });
  return admin_(auth_());
}

/* ---------------- rename sheets ---------------- */
function renameTabs_(ctx, p) {
  requireEdit_(ctx);
  const ss = SpreadsheetApp.getActive();
  (p.renames || []).forEach(r => {
    const sh = ss.getSheets().filter(x => String(x.getSheetId()) === String(r.id))[0];
    const nn = String(r.name || '').trim();
    if (!sh || !nn || sh.getName() === nn || sh.getName() === SETTINGS_SHEET) return;
    if (/[\[\]*?:\/\\]/.test(nn)) throw new Error('Sheet names can’t use [ ] * ? : / \\');
    if (ss.getSheetByName(nn)) throw new Error('There’s already a sheet called "' + nn + '"');
    sh.setName(nn);
  });
  SpreadsheetApp.flush();
  return admin_(auth_());
}

/* ---------------- fill a new year with the regular bills ---------------- */
function seriesOf_(occ) {
  // occ: [{date, amt, terms}] sorted. Works out how often it repeats.
  const ds = occ.map(o => o.date), n = ds.length;
  if (n === 1) return { every: 'year', md: ds[0].slice(5) };
  const gaps = []; for (let i = 1; i < n; i++) gaps.push(daysApart_(ds[i - 1], ds[i]));
  const share = (lo, hi) => gaps.filter(g => g >= lo && g <= hi).length / gaps.length;
  if (n >= 3 && share(6, 8) >= 0.7) return { every: 'week', days: 7 };
  if (n >= 3 && share(13, 15) >= 0.7) return { every: 'fortnight', days: 14 };
  if (n >= 3 && share(27, 29) >= 0.7 && share(30, 31) < 0.3) return { every: '4 weeks', days: 28 };
  if (n >= 3 && share(26, 34) >= 0.7) {
    const c = {}; ds.forEach(d => { const k = +d.slice(8); c[k] = (c[k] || 0) + 1; });
    let dom = 1, b = 0; Object.keys(c).forEach(k => { if (c[k] > b || (c[k] === b && +k < dom)) { b = c[k]; dom = +k; } });
    return { every: 'month', dom: dom };
  }
  if (n >= 2 && share(85, 97) >= 0.7) { return { every: 'quarter', dom: +ds[n - 1].slice(8) }; }
  return null;
}
function projectDates_(sr, last, from, to) {
  const out = [];
  if (sr.days) { let d = plusDays_(last, sr.days); while (d <= to) { if (d >= from) out.push(d); d = plusDays_(d, sr.days); } return out; }
  const y = +from.slice(0, 4);
  const at = (yy, mm, dd) => { const dim = new Date(yy, mm, 0).getDate(); return yy + '-' + ('0' + mm).slice(-2) + '-' + ('0' + Math.min(dd, dim)).slice(-2); };
  if (sr.every === 'month') { for (let m = 1; m <= 12; m++) { const d = at(y, m, sr.dom); if (d >= from && d <= to && d > last) out.push(d); } return out; }
  if (sr.every === 'quarter') { const lm = +last.slice(5, 7); for (let k = 1; k <= 8; k++) { const t = lm + 3 * k, yy = +last.slice(0, 4) + Math.floor((t - 1) / 12), mm = ((t - 1) % 12) + 1; const d = at(yy, mm, sr.dom); if (d >= from && d <= to) out.push(d); } return out; }
  if (sr.every === 'year') { const d = at(y, +sr.md.slice(0, 2), +sr.md.slice(3)); if (d >= from && d <= to) out.push(d); return out; }
  return out;
}
function planYear_(ctx, p) {
  requireEdit_(ctx);
  const X = ledgerFor_(ctx, p.tab), L = X.L;
  const year = parseInt(p.year, 10); if (!(year > 2000 && year < 2100)) throw new Error('Pick a year');
  const sFrom = (year - 1) + '-01-01', sTo = (year - 1) + '-12-31', tFrom = year + '-01-01', tTo = year + '-12-31';
  const cfg = weekdayCfg_(), wk = cfg.names.map(x => x.toLowerCase());
  const lock = LockService.getDocumentLock(); lock.waitLock(25000);
  try {
    if (!p.dryRun) extendRows_(X.sh, X.hi, L, tTo, X.tz);
    X.hi = headerInfo_(X.sh, X.hi.headerRow);
    const first = X.hi.headerRow + 1, rows = readBlock_(X, first, X.hi.lastRow - X.hi.headerRow);
    const groups = {}, existing = {};
    rows.forEach(rw => {
      if (!rw.date) return;
      const inSrc = rw.date >= sFrom && rw.date <= sTo, inTgt = rw.date >= tFrom && rw.date <= tTo;
      if (!inSrc && !inTgt) return;
      rowItems_(L, rw.pms, rw.notes).forEach(it => {
        const key = it.name ? 'n|' + it.name.toLowerCase() + '|' + it.col : 'u|' + it.col + '|' + it.amt;
        if (inSrc) { (groups[key] = groups[key] || { key: key, name: it.name, col: it.col, occ: [] }).occ.push({ date: rw.date, amt: it.amt, terms: it.terms }); }
        else (existing[key] = existing[key] || []).push(rw.date);
      });
    });
    const plan = [];
    Object.keys(groups).forEach(k => {
      const g = groups[k]; g.occ.sort((a, b) => a.date < b.date ? -1 : 1);
      const sr = seriesOf_(g.occ); if (!sr) return;
      const lastO = g.occ[g.occ.length - 1];
      const isWk = g.name && wk.indexOf(g.name.toLowerCase()) >= 0;
      let ds = projectDates_(sr, lastO.date, tFrom, tTo).map(d => isWk ? shiftWeekday_(d, cfg.dir) : d);
      const have = existing[k] || [];
      ds = ds.filter(d => !have.some(h => Math.abs(daysApart_(h, d)) <= 3));
      const already = have.length;
      if (!ds.length && !already) return;
      const colName = (L.amounts.filter(a => a.i === g.col)[0] || {}).name || '';
      const when = sr.every === 'week' || sr.every === 'fortnight' ? 'every ' + sr.every + ' on ' + DAY_NAMES[dowOf_(lastO.date)] + 's'
        : sr.every === '4 weeks' ? 'every 4 weeks' : sr.every === 'month' ? 'monthly on the ' + sr.dom + ordinal_(sr.dom)
        : sr.every === 'quarter' ? 'every 3 months' : 'once a year (' + Utilities.formatDate(dt_(lastO.date), X.tz, 'd MMM') + ')';
      // stopped part-way through the year? (e.g. a loan that was paid off) → listed, but left off
      const gap = sr.days || (sr.every === 'month' ? 31 : sr.every === 'quarter' ? 92 : 366);
      const stopped = sr.every !== 'year' && daysApart_(lastO.date, sTo) > 2 * gap + 3;
      const lastTxt = Utilities.formatDate(dt_(lastO.date), X.tz, 'd MMM');
      plan.push({ key: k, name: g.name || colName.trim(), named: !!g.name, col: g.col, colName: colName.trim(), amt: lastO.amt, terms: lastO.terms,
        every: sr.every, when: when + (stopped ? ' \u00b7 stopped ' + lastTxt : ''), count: ds.length, already: already, dates: ds, last: lastO.date,
        slot: g.col + '|' + sr.every + '|' + (sr.days ? dowOf_(lastO.date) : (sr.dom || sr.md || '')),
        pick: sr.every !== 'year' && g.occ.length >= 3 && !stopped });
    });
    // an amount with no note that changed during the year (e.g. savings $1,200 → $1,800): keep only the newest one
    const newest = {};
    plan.forEach(x => { if (x.named) return; if (!newest[x.slot] || newest[x.slot].last < x.last) newest[x.slot] = x; });
    plan.forEach(x => { if (!x.named && newest[x.slot] !== x && x.pick) { x.pick = false; x.when += ' \u00b7 replaced by ' + noteAmt_(newest[x.slot].amt); } });
    const myRep = repeatsOut_(ctx).filter(r => r.tab === X.sh.getName() && r.endMode === 'sheet');
    const repNames = {}; myRep.forEach(r => { repNames[r.name.toLowerCase() + '|' + r.col] = 1; });
    plan.forEach(x => { if (x.named && repNames[x.name.toLowerCase() + '|' + x.col]) { x.pick = false; x.when += ' \u00b7 already a repeating bill'; } });
    plan.sort((a, b) => (b.pick - a.pick) || (b.amt * b.count - a.amt * a.count));
    if (p.dryRun) return { year: year, items: plan.map(x => { const y = Object.assign({}, x); y.first = x.dates[0] || ''; delete y.dates; return y; }) };
    // write it
    const pick = {}; (p.keys || []).forEach(k => { pick[k] = true; });
    const byRow = {};
    plan.filter(x => pick[x.key]).forEach(x => x.dates.forEach(d => { (byRow[d] = byRow[d] || []).push(x); }));
    const tRows = rows.filter(rw => rw.date >= tFrom && rw.date <= tTo);
    if (!tRows.length) throw new Error('No rows for ' + year);
    const r0 = tRows[0].r, nR = tRows[tRows.length - 1].r - r0 + 1, styles = {};
    const colsOut = {}; L.amounts.concat(L.notes).forEach(c => { colsOut[c.i] = null; });
    let added = 0;
    const cellOut = (rw, ci) => rw.f[ci] ? rw.f[ci] : rw.v[ci];
    L.amounts.forEach(a => { colsOut[a.i] = tRows.map(rw => [cellOut(rw, a.i)]); });
    L.notes.forEach(nt => { colsOut[nt.i] = tRows.map(rw => [rw.v[nt.i]]); });
    tRows.forEach((rw, idx) => {
      const its = byRow[rw.date]; if (!its) return;
      its.forEach(x => {
        const a = L.amounts.filter(c => c.i === x.col)[0]; if (!a) return;
        const pm = parseMoney_(rw.f[a.i], rw.v[a.i]) || { style: null, terms: [] };
        const cur = colsOut[a.i][idx][0];
        const pm2 = typeof cur === 'string' && cur.charAt(0) === '=' ? parseMoney_(cur, '') : (cur === '' || cur == null ? { style: null, terms: [] } : parseMoney_('', cur));
        const use = pm2 || pm; if (!use) return;
        x.terms.forEach(t => use.terms.push(t));
        if (!use.style && use.terms.length > 1) use.style = styles[a.i] || (styles[a.i] = colStyle_(X.sh, X.hi, a.i));
        colsOut[a.i][idx][0] = buildMoney_(use.style, use.terms);
        if (x.named && a.note !== -1) {
          let note = String(colsOut[a.note][idx][0] || '').replace(/\s+$/, '');
          const expr = x.terms.map(noteAmt_).join('+');
          const re = new RegExp('(^|/\\s*)(' + escRe_(x.name) + ')\\s*\\(([^)]*)\\)', 'i');
          note = re.test(note) ? note.replace(re, (m, pre, nm, inner) => pre + nm + ' (' + inner + '+' + expr + ')') : (note ? note + ' / ' : '') + x.name + ' (' + expr + ')';
          colsOut[a.note][idx][0] = note;
        }
        added++;
      });
    });
    Object.keys(colsOut).forEach(ci => { if (colsOut[ci]) X.sh.getRange(r0, +ci + 1, nR, 1).setValues(colsOut[ci]); });
    SpreadsheetApp.flush();
    lock.releaseLock();
    // repeating bills with no end date carry on into the new year
    try {
      const X2 = ledgerFor_(ctx, p.tab), ents = [];
      myRep.forEach(r => rptDates(r.rule, r.start, tTo, 800).filter(d => d >= tFrom).forEach(d => ents.push({ date: d, col: r.col, name: r.name, terms: [r.amt] })));
      if (ents.length) { const lk = LockService.getDocumentLock(); lk.waitLock(25000); try { added += addEntries_(X2, ents); } finally { lk.releaseLock(); } }
    } catch (e) {}
    try { if (wk.length) weekendFix_(ctx, {}); } catch (e) {}
    return { year: year, added: added };
  } finally { try { lock.releaseLock(); } catch (e) {} }
}
function ordinal_(n) { return (n % 100 >= 11 && n % 100 <= 13) ? 'th' : ({ 1: 'st', 2: 'nd', 3: 'rd' }[n % 10] || 'th'); }


/* ============================ REPEATING BILLS ============================ */
/** Repeating dates. rule: {unit:'day'|'week'|'month'|'year', every:n, mode:'date'|'nth', nth:1-5 (5 = last), wd:0-6, wkday:true/false, dir:'after'|'before'} */
function rptDates(rule, start, until, maxN) {
  var out = [], n = Math.max(1, parseInt(rule.every, 10) || 1), cap = Math.max(1, Math.min(maxN || 800, 800));
  function pd(s) { var m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(String(s)); return new Date(+m[1], +m[2] - 1, +m[3], 12); }
  function fd(d) { return d.getFullYear() + '-' + ('0' + (d.getMonth() + 1)).slice(-2) + '-' + ('0' + d.getDate()).slice(-2); }
  function nthWd(yy, mm, nth, wd) {
    if (nth >= 5) { var last = new Date(yy, mm + 1, 0, 12); while (last.getDay() !== wd) last.setDate(last.getDate() - 1); return last; }
    var f = new Date(yy, mm, 1, 12); while (f.getDay() !== wd) f.setDate(f.getDate() + 1); f.setDate(f.getDate() + 7 * (nth - 1)); return f;
  }
  if (!/^\d{4}-\d{2}-\d{2}$/.test(String(start))) return out;
  var s = pd(start), d, ds, i;
  for (i = 0; out.length < cap && i < 3000; i++) {
    if (rule.unit === 'day' || rule.unit === 'week') { d = new Date(s); d.setDate(s.getDate() + i * n * (rule.unit === 'week' ? 7 : 1)); }
    else if (rule.unit === 'month') {
      var mo = s.getMonth() + i * n, yy = s.getFullYear() + Math.floor(mo / 12), mm = ((mo % 12) + 12) % 12;
      if (rule.mode === 'nth') d = nthWd(yy, mm, +rule.nth || 1, +rule.wd || 0);
      else d = new Date(yy, mm, Math.min(s.getDate(), new Date(yy, mm + 1, 0).getDate()), 12);
    } else { var y2 = s.getFullYear() + i * n; d = new Date(y2, s.getMonth(), Math.min(s.getDate(), new Date(y2, s.getMonth() + 1, 0).getDate()), 12); }
    ds = fd(d);
    if (until && ds > until) break;
    if (ds < start) continue;
    if (rule.wkday) { var w = d.getDay(); if (w === 6 || w === 0) { var dd = new Date(d); dd.setDate(d.getDate() + (rule.dir === 'before' ? (w === 6 ? -1 : -2) : (w === 6 ? 2 : 1))); ds = fd(dd); } }
    if (out.indexOf(ds) < 0) out.push(ds);
  }
  return out;
}

function repeatsGet_() { try { return JSON.parse(PropertiesService.getScriptProperties().getProperty('repeats') || '[]'); } catch (e) { return []; } }
function repeatsPut_(a) { PropertiesService.getScriptProperties().setProperty('repeats', JSON.stringify(a)); }
/** Repeats with the sheet's current name (follows renames by tab ID). */
function repeatsOut_(ctx) {
  const byId = {}; ctx.s.tabs.forEach(t => { byId[t.id] = t.name; });
  return repeatsGet_().map(r => { const o = Object.assign({}, r); if (r.tabId && byId[r.tabId]) o.tab = byId[r.tabId]; return o; });
}
function lastDate_(X) {
  let last = X.sh.getLastRow();
  while (last > X.hi.headerRow) { const v = X.sh.getRange(last, X.L.date + 1).getValue(); if (v instanceof Date) return ymdOf_(v, X.tz); last--; }
  return '';
}
/** Adds entries [{date, col, name, terms}] in your usual format. Skips any day that already has the same item. */
function addEntries_(X, entries) {
  if (!entries.length) return 0;
  const first = X.hi.headerRow + 1, rows = readBlock_(X, first, X.hi.lastRow - X.hi.headerRow), L = X.L;
  const byDate = {}; rows.forEach((rw, i) => { if (rw.date) byDate[rw.date] = i; });
  const touched = {}, styles = {}; let added = 0;
  entries.forEach(e => {
    const i = byDate[e.date]; if (i == null) return;
    const rw = rows[i], a = L.amounts.filter(c => c.i === e.col)[0]; if (!a) return;
    const amt = r2_(e.terms.reduce((x, y) => x + y, 0));
    const items = rowItems_(L, rw.pms, rw.notes);
    if (items.some(it => it.col === e.col && Math.abs(it.amt - amt) < 0.005 && (!e.name || it.name.toLowerCase() === e.name.toLowerCase()))) return;
    let pm = rw.pms[a.i];
    if (!pm) { if (rw.f[a.i] || String(rw.v[a.i]) !== '') return; pm = { style: null, terms: [] }; }
    e.terms.forEach(t => pm.terms.push(t));
    if (!pm.style && pm.terms.length > 1) pm.style = styles[a.i] || (styles[a.i] = colStyle_(X.sh, X.hi, a.i));
    rw.pms[a.i] = pm; (touched[a.i] = touched[a.i] || {})[i] = 1;
    if (e.name && a.note !== -1) {
      let note = String(rw.notes[a.note] || '').replace(/\s+$/, '');
      const expr = e.terms.map(noteAmt_).join('+');
      const re = new RegExp('(^|/\\s*)(' + escRe_(e.name) + ')\\s*\\(([^)]*)\\)', 'i');
      note = re.test(note) ? note.replace(re, (m, pre, nm, inner) => pre + nm + ' (' + inner + '+' + expr + ')') : (note ? note + ' / ' : '') + e.name + ' (' + expr + ')';
      rw.notes[a.note] = note; (touched[a.note] = touched[a.note] || {})[i] = 1;
    }
    added++;
  });
  // one write per column, covering just the rows that changed
  Object.keys(touched).forEach(ck => {
    const ci = +ck, idx = Object.keys(touched[ck]).map(Number), lo = Math.min.apply(null, idx), hi = Math.max.apply(null, idx);
    const isNote = L.notes.some(n => n.i === ci), out = [];
    for (let i = lo; i <= hi; i++) {
      const rw = rows[i];
      if (!touched[ck][i]) out.push([rw.f[ci] ? rw.f[ci] : rw.v[ci]]);
      else if (isNote) out.push([rw.notes[ci]]);
      else { const b = buildMoney_(rw.pms[ci].style, rw.pms[ci].terms); out.push([b]); }
    }
    X.sh.getRange(rows[lo].r, ci + 1, hi - lo + 1, 1).setValues(out);
  });
  SpreadsheetApp.flush();
  return added;
}
function repeatClean_(ctx, r) {
  const X = ledgerFor_(ctx, r.tab);
  const col = X.L.amounts.filter(a => a.i === Number(r.col))[0]; if (!col) throw new Error('Pick a column');
  const name = tidyName_(r.name || ''); if (!name) throw new Error('Give it a name (it goes in the notes)');
  const amt = r2_(r.amt); if (!(amt > 0)) throw new Error('Enter the amount');
  dt_(r.start);
  const rule = { unit: ['day', 'week', 'month', 'year'].indexOf(r.rule && r.rule.unit) >= 0 ? r.rule.unit : 'week', every: Math.max(1, Math.min(52, parseInt(r.rule && r.rule.every, 10) || 1)),
    mode: r.rule && r.rule.mode === 'nth' ? 'nth' : 'date', nth: parseInt(r.rule && r.rule.nth, 10) || 1, wd: parseInt(r.rule && r.rule.wd, 10) || 0,
    wkday: !!(r.rule && r.rule.wkday), dir: r.rule && r.rule.dir === 'before' ? 'before' : 'after' };
  const endMode = ['sheet', 'date', 'count'].indexOf(r.endMode) >= 0 ? r.endMode : 'sheet';
  const tabId = String(X.sh.getSheetId());
  return { X: X, col: col, s: { id: r.id || '', tab: X.sh.getName(), tabId: tabId, col: col.i, colName: col.name, name: name, amt: amt, start: r.start, rule: rule,
    endMode: endMode, until: endMode === 'date' ? String(r.until || '') : '', count: endMode === 'count' ? Math.max(1, Math.min(500, parseInt(r.count, 10) || 1)) : 0 } };
}
function repeatDatesFor_(X, s, from) {
  const end = lastDate_(X);
  const until = s.endMode === 'date' && s.until ? s.until : end;
  let ds = rptDates(s.rule, s.start, until, s.endMode === 'count' ? s.count : 800);
  return { all: ds, dates: ds.filter(d => d >= from), until: until, end: end };
}
function removeOcc_(ctx, s, from) {
  try { return bulkChange_(ctx, { tab: s.tab, col: s.col, oldAmt: s.amt, remove: true, from: from, name: s.name }).count; }
  catch (e) { if (/No days|Enter/.test(String(e.message))) return 0; throw e; }
}
function repeatSave_(ctx, p) {
  requireEdit_(ctx);
  const c = repeatClean_(ctx, p.series || {}), s = c.s;
  const tz = SpreadsheetApp.getActive().getSpreadsheetTimeZone(), today = ymdOf_(new Date(), tz);
  const all = repeatsGet_(), old = s.id ? all.filter(x => x.id === s.id)[0] : null;
  const from = String(p.from || '') || (old ? (s.start > today ? s.start : today) : s.start);
  let info = repeatDatesFor_(c.X, s, from);
  if (p.dryRun) return { dates: info.dates, count: info.dates.length, until: info.until, end: info.end };
  // a date past the end of the sheet: add the days first (up to 3 years ahead)
  if (s.endMode === 'date' && s.until > info.end) {
    const cap = plusDays_(today, 1100), to = s.until < cap ? s.until : cap;
    const lock = LockService.getDocumentLock(); lock.waitLock(25000);
    try { extendRows_(c.X.sh, c.X.hi, c.X.L, to, c.X.tz); } finally { lock.releaseLock(); }
  }
  let removed = 0;
  if (old) removed = removeOcc_(ctx, old, from);
  else if (p.replace) removed = removeOcc_(ctx, s, from);
  const X = ledgerFor_(ctx, s.tab);
  info = repeatDatesFor_(X, s, from);
  const lock = LockService.getDocumentLock(); lock.waitLock(25000);
  let added = 0;
  try { added = addEntries_(X, info.dates.map(d => ({ date: d, col: s.col, name: s.name, terms: [s.amt] }))); }
  finally { lock.releaseLock(); }
  if (!s.id) s.id = Utilities.getUuid().slice(0, 8);
  s.updated = today;
  const next = all.filter(x => x.id !== s.id); next.push(s); repeatsPut_(next);
  return { added: added, removed: removed, series: s, items: repeatsOut_(auth_()) };
}
function repeatDelete_(ctx, p) {
  requireEdit_(ctx);
  const all = repeatsGet_(), s = all.filter(x => x.id === p.id)[0]; if (!s) throw new Error('That repeat no longer exists');
  const tz = SpreadsheetApp.getActive().getSpreadsheetTimeZone();
  const cur = repeatsOut_(ctx).filter(x => x.id === p.id)[0] || s;
  let removed = 0;
  if (p.removeFuture !== false) removed = removeOcc_(ctx, cur, String(p.from || '') || ymdOf_(new Date(), tz));
  repeatsPut_(all.filter(x => x.id !== p.id));
  return { removed: removed, items: repeatsOut_(auth_()) };
}


/* ---------------- give an unnamed amount a name in the notes ---------------- */
function nameAmount_(ctx, p) {
  requireEdit_(ctx);
  const X = ledgerFor_(ctx, p.tab), L = X.L;
  const col = L.amounts.filter(a => a.i === Number(p.col))[0]; if (!col) throw new Error('Pick a column');
  if (col.note === -1) throw new Error('That column has no notes column beside it');
  const name = tidyName_(p.name || ''); if (!name) throw new Error('Type a name');
  const amt = r2_(p.amt); if (!(amt > 0)) throw new Error('No amount');
  const from = String(p.from || '');
  const lock = LockService.getDocumentLock(); lock.waitLock(20000);
  try {
    const first = X.hi.headerRow + 1, rows = readBlock_(X, first, X.hi.lastRow - X.hi.headerRow), hits = [];
    rows.forEach((rw, i) => {
      if (!rw.date) return;
      if (p.all ? rw.date < from : rw.r !== Number(p.row)) return;
      const items = rowItems_(L, rw.pms, rw.notes);
      if (!items.some(it => !it.name && it.col === col.i && Math.abs(it.amt - amt) < 0.005)) return;
      hits.push(i);
    });
    if (p.dryRun || !hits.length) return { count: hits.length, dates: hits.map(i => rows[i].date) };
    const lo = Math.min.apply(null, hits), hi = Math.max.apply(null, hits), out = [];
    for (let i = lo; i <= hi; i++) {
      let note = rows[i].v[col.note];
      if (hits.indexOf(i) >= 0) { const cur = String(note || '').replace(/\s+$/, ''); note = (cur ? cur + ' / ' : '') + name + ' (' + noteAmt_(amt) + ')'; }
      out.push([note]);
    }
    X.sh.getRange(rows[lo].r, col.note + 1, hi - lo + 1, 1).setValues(out);
    SpreadsheetApp.flush();
    return { count: hits.length, dates: hits.map(i => rows[i].date) };
  } finally { lock.releaseLock(); }
}
